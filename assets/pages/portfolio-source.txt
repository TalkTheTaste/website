    document.addEventListener('mouseover', e => {
        if (e.target.closest('a,button,.proj-card,.filter-btn,.media-item')) { cur.classList.add('active'); ring.classList.add('active'); }
        else { cur.classList.remove('active'); ring.classList.remove('active'); }
    });
    document.addEventListener('contextmenu', e => {
        if (e.target.closest('.protected-media,.media-item,.client-hero-media,.media-viewer-stage')) e.preventDefault();
    });
    document.addEventListener('dragstart', e => {
        if (e.target.closest('.protected-media,.media-item,.client-hero-media,.media-viewer-stage')) e.preventDefault();
    });

    let allProjects = [];
    let activeFilter = 'all';
    let portfolioSource = 'loading';
    let activeClientId = '';
    let activeClientSlug = '';
    let activeMedia = [];
    const clientDetailPromises = new Map();
    const CLIENT_CACHE_TTL = 30 * 60 * 1000;
    const PORTFOLIO_CACHE_TTL = 2 * 60 * 1000;

    const filterBar = document.getElementById('filterBar');

    function renderSkeleton() {
        filterBar.innerHTML = `<button class="filter-btn active" type="button">Loading Work</button>`;
        const grid = document.getElementById('projectsGrid');
        grid.innerHTML = Array.from({ length: 5 }).map((_, i) => `
            <div class="proj-card portfolio-skeleton ${i === 0 ? 'proj-featured' : ''}" aria-hidden="true">
                <div class="proj-thumb">
                    <div class="proj-thumb-bg"></div>
                    <div class="proj-thumb-overlay"></div>
                </div>
                <div class="proj-body">
                    <div class="proj-cat">&nbsp;</div>
                    <div class="proj-title">&nbsp;</div>
                    <div class="proj-desc">&nbsp;</div>
                    <div class="proj-tags">&nbsp;</div>
                </div>
            </div>
        `).join('');
    }

    function renderFilters() {
        const cats = ['all', ...new Set(allProjects.map(p => p.category).filter(Boolean))];
        if (!cats.includes(activeFilter)) activeFilter = 'all';
        filterBar.innerHTML = cats.map(c =>
            `<button class="filter-btn ${c === activeFilter ? 'active' : ''}" data-cat="${c}">${c === 'all' ? 'All Projects' : c}</button>`
        ).join('');
        filterBar.querySelectorAll('.filter-btn').forEach(btn => {
            btn.addEventListener('click', () => {
                filterBar.querySelectorAll('.filter-btn').forEach(b => b.classList.remove('active'));
                btn.classList.add('active');
                activeFilter = btn.dataset.cat;
                renderProjects();
            });
        });
    }

    function renderProjects() {
        const grid = document.getElementById('projectsGrid');
        const filtered = activeFilter === 'all' ? allProjects : allProjects.filter(p => p.category === activeFilter);
        if (!filtered.length) {
            grid.innerHTML = `<div class="empty-state" style="grid-column:1/-1"><div class="empty-icon">📂</div><h3>No client work yet</h3><p>Add client folders in OneDrive and they will appear here automatically.</p></div>`;
            return;
        }
        const arrowSVG = `<svg width="12" height="12" viewBox="0 0 24 24" fill="none" stroke="white" stroke-width="2" stroke-linecap="round"><path d="M7 17L17 7M7 7h10v10"/></svg>`;
        const note = portfolioSource === 'onedrive' ? `<p class="portfolio-note">Live from Talk The Taste's OneDrive portfolio.</p>` : '';
        grid.innerHTML = note + filtered.map((p, i) => {
            const thumbContent = p.image
                ? `<img src="${escapeAttr(p.image)}" alt="${escapeAttr(p.title)}" loading="lazy" style="width:100%;height:100%;object-fit:cover;display:block;">`
                : `<div class="proj-deco"><span class="proj-deco-text">${String(i+1).padStart(2,'0')}</span></div>`;
            const bgStyle = p.image
                ? 'background:#111;'
                : `background:linear-gradient(135deg,${escapeAttr(p.color1||'#0c0c24')} 0%,${escapeAttr(p.color2||'#18103c')} 60%);`;
            return `
            <div class="proj-card ${i === 0 ? 'proj-featured' : ''} reveal" data-id="${escapeAttr(p.id)}" style="transition-delay:${(i%3)*0.1}s">
                <div class="proj-thumb">
                    <div class="proj-thumb-bg" style="${bgStyle}">
                        ${thumbContent}
                    </div>
                    <div class="proj-thumb-overlay"></div>
                    <div class="proj-arrow">${arrowSVG}</div>
                </div>
                <div class="proj-body">
                    <div class="proj-cat">${escapeHtml(p.category || '')}</div>
                    <div class="proj-title">${escapeHtml(p.title)}</div>
                <div class="proj-desc">${escapeHtml(p.description || '')}</div>
                <div class="proj-tags">${(p.tags||[]).map(t => `<span class="proj-tag">${escapeHtml(t)}</span>`).join('')}</div>
                </div>
            </div>`;
        }).join('');
        grid.querySelectorAll('.reveal').forEach(el => obs.observe(el));
        grid.querySelectorAll('.proj-card').forEach(card => {
            card.addEventListener('click', () => openModal(card.dataset.id));
        });
    }

    function refreshProjectData() {
        if (portfolioSource === 'onedrive' || portfolioSource === 'loading') return;
        allProjects = TTT.getProjects();
        portfolioSource = 'local';
        renderFilters();
        renderProjects();
    }

    hydratePortfolioCache();
    loadOneDrivePortfolio();
    document.addEventListener('ttt:data:ready', refreshProjectData);

    let pendingClientParam = new URLSearchParams(location.search).get('client') || new URLSearchParams(location.search).get('id') || '';
    if (pendingClientParam) setTimeout(() => openClientFromParam(pendingClientParam, true), 600);

    async function openModal(id, replaceUrl = false) {
        let p = allProjects.find(project => project.id === id);
        if (!p) return;
        activeClientId = id;
        activeClientSlug = clientSlug(p);
        updateClientUrl(p, replaceUrl);
        renderClientView(p, portfolioSource === 'onedrive' && !p.media?.length);
        document.getElementById('clientView').classList.add('open');
        document.getElementById('clientView').setAttribute('aria-hidden', 'false');
        document.body.style.overflow = 'hidden';
        if (portfolioSource === 'onedrive' && !p.media?.length) {
            try {
                const detail = await loadClientDetail(id);
                p = { ...p, ...detail };
                allProjects = allProjects.map(project => project.id === id ? p : project);
                if (activeClientId === id && document.getElementById('clientView').classList.contains('open')) renderClientView(p, false);
            } catch (error) {
                document.getElementById('clientGallery').innerHTML = `<div class="portfolio-loading">This client gallery could not be loaded right now.</div>`;
            }
        }
    }

    async function loadOneDrivePortfolio() {
        const grid = document.getElementById('projectsGrid');
        try {
            const response = await fetch('/api/portfolio', { headers: { Accept: 'application/json' } });
            if (!response.ok) throw new Error('Portfolio request failed');
            const data = await response.json();
            if (!Array.isArray(data.projects) || data.source !== 'onedrive') throw new Error('OneDrive portfolio unavailable');
            allProjects = data.projects;
            portfolioSource = data.source;
            setPortfolioCache(data);
            renderFilters();
            renderProjects();
            prefetchClientDetails();
            if (pendingClientParam) {
                openClientFromParam(pendingClientParam, true);
                pendingClientParam = '';
            }
        } catch (error) {
            if (!allProjects.length) {
                portfolioSource = 'local';
                allProjects = TTT.getProjects();
                if (allProjects.length) {
                    renderFilters();
                    renderProjects();
                } else {
                    grid.innerHTML = `<div class="empty-state" style="grid-column:1/-1"><div class="empty-icon">📂</div><h3>Portfolio unavailable</h3><p>Client work could not be loaded right now.</p></div>`;
                }
            }
        }
    }

    function hydratePortfolioCache() {
        const cached = getPortfolioCache();
        if (cached?.projects?.length) {
            allProjects = cached.projects;
            portfolioSource = 'onedrive';
            renderFilters();
            renderProjects();
            prefetchClientDetails();
            return;
        }
        renderSkeleton();
    }

    function getPortfolioCache() {
        try {
            const raw = localStorage.getItem('ttt_portfolio_index');
            if (!raw) return null;
            const cached = JSON.parse(raw);
            if (!cached || Date.now() - cached.savedAt > PORTFOLIO_CACHE_TTL) return null;
            return cached.data;
        } catch {
            return null;
        }
    }

    function setPortfolioCache(data) {
        try {
            localStorage.setItem('ttt_portfolio_index', JSON.stringify({ savedAt: Date.now(), data }));
        } catch {}
    }

    function openClientFromParam(param, replaceUrl = false) {
        const target = String(param || '').trim();
        if (!target) return;
        const project = allProjects.find(item => item.id === target || clientSlug(item) === target);
        if (project) openModal(project.id, replaceUrl);
    }

    function prefetchClientDetails() {
        if (portfolioSource !== 'onedrive') return;
        const clients = allProjects.filter(project => project.id && !project.media?.length).slice(0, 10);
        let index = 0;
        const runNext = () => {
            const project = clients[index++];
            if (!project) return;
            loadClientDetail(project.id)
                .then(detail => {
                    allProjects = allProjects.map(item => item.id === project.id ? { ...item, ...detail } : item);
                    if (activeClientId === project.id && document.getElementById('clientView').classList.contains('open')) {
                        renderClientView(allProjects.find(item => item.id === project.id), false);
                    }
                })
                .catch(() => {})
                .finally(() => setTimeout(runNext, 350));
        };
        const start = () => {
            runNext();
            setTimeout(runNext, 220);
        };
        if ('requestIdleCallback' in window) requestIdleCallback(start, { timeout: 1400 });
        else setTimeout(start, 800);
    }

    function loadClientDetail(id) {
        const cached = getClientCache(id);
        if (cached) return Promise.resolve(cached);
        if (clientDetailPromises.has(id)) return clientDetailPromises.get(id);
        const promise = fetch(`/api/portfolio/client?id=${encodeURIComponent(id)}`, { headers: { Accept: 'application/json' } })
            .then(response => {
                if (!response.ok) throw new Error('Client gallery request failed');
                return response.json();
            })
            .then(detail => {
                setClientCache(id, detail);
                return detail;
            })
            .finally(() => clientDetailPromises.delete(id));
        clientDetailPromises.set(id, promise);
        return promise;
    }

    function getClientCache(id) {
        try {
            const raw = sessionStorage.getItem(`ttt_client_${id}`);
            if (!raw) return null;
            const cached = JSON.parse(raw);
            if (!cached || Date.now() - cached.savedAt > CLIENT_CACHE_TTL) return null;
            return cached.detail;
        } catch {
            return null;
        }
    }

    function setClientCache(id, detail) {
        try {
            sessionStorage.setItem(`ttt_client_${id}`, JSON.stringify({ savedAt: Date.now(), detail }));
        } catch {}
    }

    function renderClientView(project, loading) {
        activeMedia = Array.isArray(project.media) ? project.media : [];
        document.getElementById('clientHeroMedia').innerHTML = project.image
            ? `<img class="protected-media" src="${escapeAttr(project.image)}" alt="${escapeAttr(project.title)}" draggable="false">`
            : '';
        document.getElementById('clientKicker').textContent = project.category || 'Client Work';
        document.getElementById('clientTitle').textContent = project.title;
        document.getElementById('clientDesc').textContent = project.description || '';
        document.getElementById('clientStatus').textContent = loading ? 'Loading gallery...' : cleanCountText(project, activeMedia);
        document.getElementById('clientGallery').innerHTML = loading
            ? `<div class="portfolio-loading">Loading ${escapeHtml(project.title)} gallery...</div>`
            : projectContent(project);
    }

    function projectContent(project) {
        const media = Array.isArray(project.media) ? project.media : [];
        if (!media.length) return project.fullDesc || `<p>${escapeHtml(project.description || '')}</p>`;
        return `
            <div class="media-gallery">
                ${media.map((item, index) => mediaItem(item, project.title, mediaIndex(media, item, index), index)).join('')}
            </div>
        `;
    }

    function mediaItem(item, clientTitle, cleanIndex, index) {
        const isVideo = item.type === 'Videography' || /^video\//i.test(item.mimeType || '');
        const label = `${clientTitle} ${isVideo ? 'Video' : 'Photo'} ${cleanIndex}`;
        return `<button type="button" class="media-item ${isVideo ? 'media-video' : 'media-photo'}" data-media-index="${index}" aria-label="${escapeAttr(label)}" style="animation-delay:${Math.min(index, 18) * 0.025}s">
            <img class="protected-media" src="${escapeAttr(item.thumb || item.url)}" alt="${escapeAttr(label)}" loading="lazy" draggable="false">
            <span class="media-label">${escapeHtml(isVideo ? `Video ${cleanIndex}` : `Photo ${cleanIndex}`)}</span>
        </button>`;
    }

    function mediaIndex(media, item, index) {
        const isVideo = item.type === 'Videography' || /^video\//i.test(item.mimeType || '');
        return media.slice(0, index + 1).filter(current => {
            const currentIsVideo = current.type === 'Videography' || /^video\//i.test(current.mimeType || '');
            return currentIsVideo === isVideo;
        }).length;
    }

    function cleanCountText(project, media) {
        const photos = media.filter(item => !(item.type === 'Videography' || /^video\//i.test(item.mimeType || ''))).length;
        const videos = media.length - photos;
        if (photos || videos) return [photos ? `${photos} photo${photos === 1 ? '' : 's'}` : '', videos ? `${videos} video${videos === 1 ? '' : 's'}` : ''].filter(Boolean).join(' and ');
        return project.description || '';
    }

    function openMediaViewer(index) {
        const item = activeMedia[index];
        if (!item) return;
        const isVideo = item.type === 'Videography' || /^video\//i.test(item.mimeType || '');
        const stage = document.getElementById('mediaViewerStage');
        if (isVideo) {
            stage.innerHTML = `<video class="protected-media" src="${escapeAttr(item.url)}" poster="${escapeAttr(item.thumb || '')}" controls controlsList="nodownload noremoteplayback" disablepictureinpicture playsinline autoplay draggable="false"></video>`;
            const video = stage.querySelector('video');
            video.addEventListener('loadedmetadata', () => {
                video.classList.toggle('is-portrait', video.videoHeight > video.videoWidth);
                video.classList.toggle('is-landscape', video.videoWidth >= video.videoHeight);
            }, { once:true });
        } else {
            stage.innerHTML = `<img class="protected-media" src="${escapeAttr(item.url)}" alt="" draggable="false">`;
        }
        document.getElementById('mediaViewer').classList.add('open');
        document.getElementById('mediaViewer').setAttribute('aria-hidden', 'false');
        document.body.style.overflow = 'hidden';
    }

    function closeMediaViewer() {
        document.getElementById('mediaViewer').classList.remove('open');
        document.getElementById('mediaViewer').setAttribute('aria-hidden', 'true');
        document.getElementById('mediaViewerStage').innerHTML = '';
        document.body.style.overflow = document.getElementById('clientView').classList.contains('open') ? 'hidden' : '';
    }

    function escapeHtml(value) {
        return String(value || '').replace(/[&<>"']/g, char => ({
            '&': '&amp;',
            '<': '&lt;',
            '>': '&gt;',
            '"': '&quot;',
            "'": '&#39;',
        }[char]));
    }

    function escapeAttr(value) {
        return escapeHtml(value);
    }

    function closeModal(updateUrl = true) {
        closeMediaViewer();
        document.getElementById('clientView').classList.remove('open');
        document.getElementById('clientView').setAttribute('aria-hidden', 'true');
        activeClientId = '';
        activeClientSlug = '';
        activeMedia = [];
        document.body.style.overflow = '';
        if (updateUrl && (new URLSearchParams(location.search).has('client') || new URLSearchParams(location.search).has('id'))) {
            history.pushState({}, '', location.pathname);
        }
    }

    document.getElementById('clientClose').addEventListener('click', closeModal);
    document.getElementById('clientShare').addEventListener('click', copyClientLink);
    document.getElementById('mediaViewerClose').addEventListener('click', closeMediaViewer);
    document.getElementById('mediaViewer').addEventListener('click', e => { if(e.target === document.getElementById('mediaViewer')) closeMediaViewer(); });
    document.getElementById('clientGallery').addEventListener('click', e => {
        const item = e.target.closest('.media-item');
        if (item) openMediaViewer(Number(item.dataset.mediaIndex));
    });
    document.addEventListener('keydown', e => {
        if (e.key !== 'Escape') return;
        if (document.getElementById('mediaViewer').classList.contains('open')) closeMediaViewer();
        else closeModal();
    });
    window.addEventListener('popstate', () => {
        const param = new URLSearchParams(location.search).get('client') || new URLSearchParams(location.search).get('id') || '';
        if (param) openClientFromParam(param, true);
        else if (document.getElementById('clientView').classList.contains('open')) closeModal(false);
    });

    async function copyClientLink() {
        if (!activeClientSlug) return;
        const url = `${location.origin}${location.pathname}?client=${encodeURIComponent(activeClientSlug)}`;
        try {
            await navigator.clipboard.writeText(url);
            setShareLabel('Copied');
        } catch {
            const input = document.createElement('input');
            input.value = url;
            input.style.position = 'fixed';
            input.style.opacity = '0';
            document.body.appendChild(input);
            input.select();
            document.execCommand('copy');
            input.remove();
            setShareLabel('Copied');
        }
    }

    function setShareLabel(text) {
        const label = document.querySelector('#clientShare span');
        if (!label) return;
        const original = 'Share';
        label.textContent = text;
        clearTimeout(setShareLabel.timer);
        setShareLabel.timer = setTimeout(() => { label.textContent = original; }, 1600);
    }

    function updateClientUrl(project, replace = false) {
        const url = `${location.pathname}?client=${encodeURIComponent(clientSlug(project))}`;
        if (replace) history.replaceState({}, '', url);
        else history.pushState({}, '', url);
    }

    function clientSlug(project) {
        return String(project.title || project.id || '')
            .toLowerCase()
            .replace(/&/g, ' and ')
            .replace(/[^a-z0-9]+/g, '-')
            .replace(/^-|-$/g, '') || String(project.id || '');
    }

})();
